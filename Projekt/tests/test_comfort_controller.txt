import sys
import os

sys.path.append(os.path.join(os.path.dirname(__file__), "..", "src", "backend"))

from comfort_controller import ComfortProfile, analyze_comfort


def test_high_co2_turns_on_ventilation():
    profile = ComfortProfile(max_co2=1000.0)

    measurement = {
        "temperature": 22.0,
        "humidity": 45.0,
        "co2": 1500.0,
        "light_level": 400.0,
        "noise_level": 40.0,
        "occupancy": True
    }

    decision = analyze_comfort(measurement, profile)

    assert decision["ventilation"] == "ON"
    assert decision["alert"] == "Przekroczono dopuszczalny poziom CO2."


def test_empty_room_enables_eco_mode():
    profile = ComfortProfile()

    measurement = {
        "temperature": 22.0,
        "humidity": 45.0,
        "co2": 700.0,
        "light_level": 500.0,
        "noise_level": 35.0,
        "occupancy": False
    }

    decision = analyze_comfort(measurement, profile)

    assert decision["mode"] == "ECO"
    assert decision["ventilation"] == "ECO"
    assert decision["lighting"] == "OFF"
